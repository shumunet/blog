## 前言

树链剖分是解决树上在线问题的一个很好用的技巧，可以在对数复杂度下维护一些问题。学习本文前，请务必确保已经掌握了线段树、LCA、树上差分和树上前缀和、dfs 序的知识点。

## 基本概念

对于树剖这一技巧具体如何剖分一颗树。我们由浅入深，逐步来讲解。

- 树大小：某棵树中含有的元素数目。

- 重儿子：某个父节点的儿子中，子树大小最大的节点。（特别规定：一个节点的重儿子只能有一个）
- 轻儿子：某个父节点的儿子中，非重儿子的节点。
- 重边：父节点和它的重儿子连成的边。
- 轻边：父节点和它的轻儿子连成的边。
- 重链：由多条重边连成的链。

## 核心思想

树链剖分的核心思想是把路径变成区间。线段树只能处理**数组区间**。树上的路径是弯的，怎么办？

树剖做了下面的两件事：

1. 给每个点选一个**重儿子**，把树分成若干条**重链**（一条从上往下、全由重儿子连成的链）。
2. 按“先重儿子”的 DFS 序给点重新编号 `dfn`。这样得到两个关键性质：
   1. 同一条重链上的点，`dfn` 是连续的。
   2. 任意一棵子树，`dfn` 也是连续的。

根据性质1，路径会被拆分成 $O(\log n)$ 段区间，根据性质2，子树报错变成了一次区间操作。

## 详细实现

接下来我们结合具体代码说明树剖的详细实现，通常情况下，我们使用两次 DFS 来实现树剖。

### 变量含义

- $siz[u]$ 子树的大小。
- $son[u]$ 重儿子。
- $top[u]$ 所在重链的链头（最上面的那个点）。
- $dep[u]$ 深度。
- $fa[u]$ 父亲节点。
- $dfn[u]$ 表示在序列上节点 $u$ 的编号。
- $rk[u]$ 表示新序列上第 $u$ 个数的数值。

### 第一次 DFS

第一次 DFS 主要是用于维护 `fa,dep,siz,son` 的，我们直接贴出代码实现。

```c++
void dfs1(int u,int f) {
    fa[u]=f,dep[u]=dep[f]+1,siz[u]=1;
    for(int v:edge[u]) {
        if(v==f) continue;
        dfs1(v,u);
        siz[u]+=siz[v];
        if(siz[v]>siz[son[u]]) son[u]=v;
    }
}
```

### 第二次 DFS

第二次 DFS 的核心是**先递归重儿子，再递归轻儿子**，它是用来分配 $dfn$ 的，其中传入的 $t$ 代表链头。

```c++
void dfs2(int u,int t) {
    top[u]=t,dfn[u]=++cnt,rk[cnt]=u;
    if(!son[u]) return;
    dfs2(son[u],t);
    //重儿子继承链头
    for(int v:edge[u]) {
        if(v!=fa[u]&&v!=son[u]) dfs2(v,v); 
        //轻儿子自己当链头 
    }
}
```

### 跳链（重点）

求路径 $u \rightarrow v$，路径一定是这样的形状：

```
        LCA
       /   \
   (上坡)   (下坡)
     /       \
    u         v
```

而在线段树中，只能区间求和不能下坡，所以说我们把右边那段倒过来看，变成 $v$ 往上爬到 $LCA$。

于是整条路径就变成了 $u \rightarrow LCA$ 和 $v \rightarrow LCA$ 两段，最后在 $LCA$ 处接上。

> 跳链的本质：让 $u$ 和 $v$ 各自往上爬，每次只爬当前这条链，从链头到自己的这一段，爬完换到上面一条链，直到两人在同一条链上。

为什么一次只能爬一段呢？因为只有同一条链上的连续点才满足 $dfn$ 连续这一性质，才能一次性丢给线段树，跨链的话 $dfn$ 是断开的，必须分次。

```c++
void path_add(int u,int v,int k) {
    while (top[u] != top[v]) {                          // ① 还在不同链上，就继续爬
        if (dep[top[u]] < dep[top[v]]) swap(u, v);      // ② 谁的链头深，就让谁爬
        update(1, 1, n, dfn[top[u]], dfn[u], k);        // ③ 把这段连续 dfn 交给线段树
        u = fa[top[u]];                                 // ④ 爬过链头，换到上面那条链
    }
    if (dep[u] > dep[v]) swap(u, v);                    // ⑤ 同链了，浅的那个就是 LCA
    update(1, 1, n, dfn[u], dfn[v], k);                 // ⑥ 把最后一段也处理掉
}
```

下面我们来分析跳链的代码：

- ① : `top[u]` 是 $u$ 所在的链头。`top[u]==top[v]` 意味着二者在同一条链上，这时路径就是链上的一段，$dfn$ 连续，直接给线段树。
- ② : 让“链头更深”的那个人站着，让“链头更浅”的那个人去爬。原因：保证**砍下来的这一段绝不会越过 LCA**，这样才不会加多。
- ③ : 线段树更新区间（同一条链 $dfn$ 连续）。
- ④ : 爬过链头，换到上面那条链。
- ⑤⑥ : 退出循环的时候 `top[u]==top[v]` ，两人在同一条链上。这条链上从 u 到 v 的那一段就是路径的**最后一段**，而且**深度小的那个就是 LCA**，所以 `swap` 保证 u 是浅的，然后处理 `[dfn[u], dfn[v]]`。

查询的话和它结构式类似的，这里直接贴出代码：

```c++
int path_query(int u,int v) {
    int res=0;
    while(top[u]!=top[v]) {
        if(dep[top[u]<dep[top[v]]]) swap(u,v);
        res=(res+query(1,1,n,dfn[top[u]],dfn[u]))%p;
        u=fa[top[u]];
    }
    if(dep[u]>dep[v]) swap(u,v);
    res=(res+query(1,1,n,dfn[u],dfn[v]))%p;
    return res;
}
```

至此，我们已经完成了树链剖分部分的代码。

## [模板题](https://www.luogu.com.cn/problem/P3384)

就是普通的区间加与区间求和，使用懒标记线段树即可维护，唯一要注意的就是它是建立在 $dfn$ 上的线段树，建树的时候要记得取 $dfn$ 而不是原始的 $a$。

```c++
#include<bits/stdc++.h>
using namespace std;
#define int long long
const int MAXN = 2e5+100;
#define endl '\n'

vector<int> edge[MAXN];
int n,m,r,mod,cnt;
int siz[MAXN],fa[MAXN],son[MAXN],dep[MAXN],top[MAXN],dfn[MAXN],rk[MAXN];
int tree[MAXN<<2],lzy[MAXN<<2],a[MAXN],w[MAXN];

void push_up(int p) {
    tree[p]=(tree[p<<1]+tree[p<<1|1])%mod;
}

void push_down(int p,int l,int r) {
    if(!lzy[p]) return;
    int mid=(l+r)>>1;
    tree[p<<1]+=(mid-l+1)*lzy[p];
    tree[p<<1]%=mod;
    lzy[p<<1]+=lzy[p];
    tree[p<<1|1]+=(r-mid)*lzy[p];
    tree[p<<1|1]%=mod;
    lzy[p<<1|1]+=lzy[p];
    lzy[p]=0;
}

void build(int p,int l,int r) {
    if(l==r) {
        tree[p]=w[l]%mod;
        return;
    }
    int mid=(l+r)>>1;
    build(p<<1,l,mid);
    build(p<<1|1,mid+1,r);
    push_up(p);
}

void update(int p,int l,int r,int ql,int qr,int k) {
    if(ql<=l&&qr>=r) {
        tree[p]=(tree[p]+k*(r-l+1))%mod;
        lzy[p]=(lzy[p]+k)%mod;
        return;
    }
    int mid=(l+r)>>1;
    push_down(p,l,r);
    if(ql<=mid) update(p<<1,l,mid,ql,qr,k);
    if(qr>mid) update(p<<1|1,mid+1,r,ql,qr,k);
    push_up(p);
}

int query(int p,int l,int r,int ql,int qr) {
    if(ql<=l&&qr>=r) {
        return tree[p];
    }
    push_down(p,l,r);
    int mid=(l+r)>>1;
    int res=0;
    if(ql<=mid) res+=query(p<<1,l,mid,ql,qr),res%=mod;
    if(qr>mid) res+=query(p<<1|1,mid+1,r,ql,qr),res%=mod;
    return res%mod;
}

void dfs1(int u,int f) {
    fa[u]=f,dep[u]=dep[f]+1,siz[u]=1;
    for(int v:edge[u]) {
        if(v==f) continue;
        dfs1(v,u);
        siz[u]+=siz[v];
        if(siz[v]>siz[son[u]]) son[u]=v;
    }
}

void dfs2(int u,int t) {
    top[u]=t,dfn[u]=++cnt,rk[cnt]=u;
    if(!son[u]) return;
    dfs2(son[u],t);
    //重儿子继承链头
    for(int v:edge[u]) {
        if(v!=fa[u]&&v!=son[u]) dfs2(v,v); 
        //轻儿子自己当链头 
    }
}

void path_add(int u,int v,int k) {
    while (top[u] != top[v]) {                          // ① 还在不同链上，就继续爬
        if (dep[top[u]] < dep[top[v]]) swap(u, v);      // ② 谁的链头深，就让谁爬
        update(1, 1, n, dfn[top[u]], dfn[u], k);        // ③ 把这段连续 dfn 交给线段树
        u = fa[top[u]];                                 // ④ 爬过链头，换到上面那条链
    }
    if (dep[u] > dep[v]) swap(u, v);                    // ⑤ 同链了，浅的那个就是 LCA
    update(1, 1, n, dfn[u], dfn[v], k);                 // ⑥ 把最后一段也处理掉
}

int path_query(int u,int v) {
    int res=0;
    while(top[u]!=top[v]) {
        if(dep[top[u]]<dep[top[v]]) swap(u,v);
        res=(res+query(1,1,n,dfn[top[u]],dfn[u]))%mod;
        u=fa[top[u]];
    }
    if(dep[u]>dep[v]) swap(u,v);
    res=(res+query(1,1,n,dfn[u],dfn[v]))%mod;
    return res;
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin>>n>>m>>r>>mod;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<n;i++) {
        int u,v;
        cin>>u>>v;
        edge[u].push_back(v);
        edge[v].push_back(u);
    }
    dfs1(r,0);
    dfs2(r,r);
    for(int i=1;i<=n;i++) w[i]=a[rk[i]];
    build(1,1,n);
    while(m--) {
        int op;
        cin>>op;
        if(op==1) {
            int x,y,z;
            cin>>x>>y>>z;
            path_add(x,y,z);
        }
        else if(op==2) {
            int x,y;
            cin>>x>>y;
            cout<<path_query(x,y)<<endl;
        }
        else if(op==3) {
            int x,z;
            cin>>x>>z;
            update(1,1,n,dfn[x],dfn[x]+siz[x]-1,z);
        }
        else {
            int x;
            cin>>x;
            cout<<query(1,1,n,dfn[x],dfn[x]+siz[x]-1)<<endl;
        }
    }
    return 0;
}
```

## [[NOI2015] 软件包管理器](https://www.luogu.com.cn/problem/P2146)

跟模板题差不多，软件包安装器本身的依赖关系构成了一棵树，用 0 代表没安装，用 1 代表安装了，然后用线段树维护区间安装的数目就行了。有个实现技巧就是这道题是 0 开始编号的，我们可以给虚拟编号搞成 -1。而用 0 代表没安装我们可以把没有懒标记搞成 -1。

```c++
#include<bits/stdc++.h>
using namespace std;
const int MAXN = 2e5+100;
#define endl '\n'
#define int long long

int n;
int tree[MAXN<<2],lzy[MAXN<<2];
int siz[MAXN],top[MAXN],son[MAXN],fa[MAXN],dep[MAXN],cnt,rk[MAXN],dfn[MAXN],w[MAXN];
vector<int> edge[MAXN];

void push_up(int p) {
    tree[p]=tree[p<<1]+tree[p<<1|1];
}

void push_down(int p,int l,int r) {
    if(lzy[p]==-1) return;
    int mid=(l+r)>>1;
    tree[p<<1]=(lzy[p])*(mid-l+1);
    tree[p<<1|1]=(lzy[p])*(r-mid);
    lzy[p<<1]=lzy[p];
    lzy[p<<1|1]=lzy[p];
    lzy[p]=-1;
}

void build(int p,int l,int r) {
    if(l==r) {
        tree[p]=w[l];
        return;
    }
    int mid=(l+r)>>1;
    build(p<<1,l,mid);
    build(p<<1|1,mid+1,r);
    push_up(p);
}

void update(int p,int l,int r,int ql,int qr,int k) {
    if(ql<=l&&qr>=r) {
        tree[p]=k*(r-l+1);
        lzy[p]=k;
        return;
    }
    push_down(p,l,r);
    int mid=(l+r)>>1;
    if(ql<=mid) update(p<<1,l,mid,ql,qr,k);
    if(qr>mid) update(p<<1|1,mid+1,r,ql,qr,k);
    push_up(p);
}

int query(int p,int l,int r,int ql,int qr) {
    if(ql<=l&&qr>=r) {
        return tree[p];
    }
    push_down(p,l,r);
    int res=0;
    int mid=(l+r)>>1;
    if(ql<=mid) res+=query(p<<1,l,mid,ql,qr);
    if(qr>mid) res+=query(p<<1|1,mid+1,r,ql,qr);
    return res;
}

void dfs1(int u,int f) {
    dep[u]=dep[f]+1,siz[u]=1,fa[u]=f;
    for(int v:edge[u]) {
        if(v==f) continue;
        dfs1(v,u);
        siz[u]+=siz[v];
        if(son[u]==-1||siz[v]>siz[son[u]]) son[u]=v;
    }
}

void dfs2(int u,int t) {
    top[u]=t,dfn[u]=++cnt,rk[cnt]=u;
    if(son[u]==-1) return;
    dfs2(son[u],t);
    for(int v:edge[u]) {
        if(v!=fa[u]&&v!=son[u]) dfs2(v,v);
    }
}

void path_update(int u,int v,int k) {
    while(top[u]!=top[v]) {
        if(dep[top[u]]<dep[top[v]]) swap(u,v);
        update(1,1,n,dfn[top[u]],dfn[u],k);
        u=fa[top[u]];
    }
    if(dep[u]>dep[v]) swap(u,v);
    update(1,1,n,dfn[u],dfn[v],k);
}

int path_query(int u,int v) {
    int res=0;
    while(top[u]!=top[v]) {
        if(dep[top[u]]<dep[top[v]]) swap(u,v);
        res+=query(1,1,n,dfn[top[u]],dfn[u]);
        u=fa[top[u]];
    }
    if(dep[u]>dep[v]) swap(u,v);
    res+=query(1,1,n,dfn[u],dfn[v]);
    return res;
}

void install(int x) {
    int len=dep[x];
    int old=path_query(0,x);
    cout<<len-old<<endl;
    path_update(0,x,1);
}

void uninstall(int x) {
    int old=query(1,1,n,dfn[x],dfn[x]+siz[x]-1);
    cout<<old<<endl;
    update(1,1,n,dfn[x],dfn[x]+siz[x]-1,0);
}

signed main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin>>n;
    for(int i=1;i<n;i++) {
        int v;
        cin>>v;
        edge[i].push_back(v);
        edge[v].push_back(i);
    }
    memset(son,-1,sizeof(son));
    memset(lzy,-1,sizeof(lzy));
    dfs1(0,0);
    dfs2(0,0);
    build(1,1,n);
    int q;
    cin>>q;
    while(q--) {
        string op;
        int x;
        cin>>op>>x;
        if(op=="install") {
            install(x);
        }
        else {
            uninstall(x);
        }
    }
    return 0;
}
```
